food_bill = float(input("Enter Food Bill: "))

gst = food_bill * 0.18
service_charge = food_bill * 0.05

total = food_bill + gst + service_charge

print("GST =", gst)
print("Service Charge =", service_charge)
print("Total Bill =", total)

1000
Sample Output
GST = 180.0
Service Charge = 50.0
