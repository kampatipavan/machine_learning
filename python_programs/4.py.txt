pin = input("Enter PIN: ")

balance = float(input("Enter Account Balance: "))
amount = float(input("Enter Withdrawal Amount: "))

if pin == "1234":
    if amount <= balance:
        balance = balance - amount
        print("Transaction Successful")
        print("Remaining Balance =", balance)
    else:
        print("Insufficient Balance")
else:
    print("Invalid PIN")
