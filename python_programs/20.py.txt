Python Code
tickets = int(input("Enter Number of Tickets: "))
category = input("Enter Seat Category (Silver/Gold): ")

if category.lower() == "silver":
    price = 150
elif category.lower() == "gold":
    price = 250
else:
    price = 0

total = tickets * price

if tickets >= 5:
    total = total * 0.9

print("Total Ticket Cost =", total)

5
Gold
